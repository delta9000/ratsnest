**Goal:** Implement Phase 2's `adsr` keyboard envelope — func's sibling with a gate input and a live-tracking sustain stage — plus load-bearing migration coverage and a demo patch.

**Architecture:** One module file per the established pattern; the stage machine, `curve` config, `increment()`/`shape()` helpers, and double accumulators mirror `src/dsp/modules/func.cpp` deliberately. Design rationale: `docs/superpowers/plans/2026-07-11-adsr-design.md`.

**Tech Stack:** C++20, doctest, CMake globs. `mise run build` / `mise run test`.

## Global Constraints

- `Module::process()` never allocates, locks, or does I/O; `copy_state_from()` is RT-called, same rule.
- Signal conventions (spec §4): gate rises through 0.5; envelope output unipolar 0..1; times in seconds (`ms` sugar).
- `progress_`/`level_` accumulate in **double** — func froze past 2^25 samples in float (fixed in ee97a4a-era review); do not reintroduce.
- Migration tests must be load-bearing (mutation-verified); the held-gate trap documented in `tests/test_migrate.cpp`'s comments applies directly here.
- Docs are part of the diff: spec Phase 2 adsr row updated in Task 1's commit.
- Conventional commits, no AI attribution. No new deps.

**Reference implementation:** read `src/dsp/modules/func.cpp` first — adsr reuses its idioms verbatim (kIns/kCfg tables, `cfg_str` curve parse, `increment`, `shape`, capture-current-level stage entry).

```cpp
#include <doctest/doctest.h>
#include <string>
#include "dsp/modules.h"

using namespace ms::dsp;

namespace {
struct AdsrRig {
  std::unique_ptr<Module> module;
  // gate, attack, decay, sustain, release
  float ins[5] = {0.0f, 0.01f, 0.02f, 0.5f, 0.04f};

  explicit AdsrRig(const char* curve = "lin") {
    const ModuleInfo* info = find_module("adsr");
    REQUIRE(info != nullptr);
    module = info->create();
    Config cfg;
    ConfigValue v; v.str = curve;
    cfg.push_back({"curve", v});
    module->init(1000.0, cfg);  // 1 kHz: attack 10 samples, decay 20, release 40
  }

  float tick(float gate) {
    ins[0] = gate;
    float out[1];
    module->process(ins, out);
    return out[0];
  }
};
}  // namespace

TEST_CASE("attack reaches one in exactly attack seconds, then decays to sustain") {
  AdsrRig env;
  float peak = 0.0f;
  int peak_at = -1;
  for (int i = 0; i < 10; ++i) {
    const float o = env.tick(1.0f);
    if (o > peak) { peak = o; peak_at = i; }
  }
  CHECK(peak == doctest::Approx(1.0f));
  CHECK(peak_at == 9);  // 10 samples of attack at 1 kHz with 0.01 s
  for (int i = 0; i < 20; ++i) (void)env.tick(1.0f);
  CHECK(env.tick(1.0f) == doctest::Approx(0.5f));  // settled on sustain
}

TEST_CASE("sustain tracks its input live while the gate is high") {
  AdsrRig env;
  for (int i = 0; i < 40; ++i) (void)env.tick(1.0f);  // well into sustain
  env.ins[3] = 0.8f;
  CHECK(env.tick(1.0f) == doctest::Approx(0.8f));
  env.ins[3] = 0.2f;
  CHECK(env.tick(1.0f) == doctest::Approx(0.2f));
}

TEST_CASE("release runs from the current level when the gate falls") {
  AdsrRig env;
  for (int i = 0; i < 40; ++i) (void)env.tick(1.0f);  // sustain 0.5
  const float first = env.tick(0.0f);
  CHECK(first < 0.5f);
  CHECK(first > 0.45f);  // one release step from 0.5, not a jump
  for (int i = 0; i < 45; ++i) (void)env.tick(0.0f);

TEST_CASE("a gate blip shorter than the attack releases from partway up") {

  for (int i = 0; i < 5; ++i) top = env.tick(1.0f);  // half the attack
  CHECK(top == doctest::Approx(0.5f));
  const float after = env.tick(0.0f);  // released mid-attack
  CHECK(after < top);
  CHECK(after > 0.0f);
  float last = after;
  for (int i = 0; i < 60; ++i) last = env.tick(0.0f);
  CHECK(last == doctest::Approx(0.0f));

TEST_CASE("retrigger during release restarts the attack from the current level") {
  AdsrRig env;
  for (int i = 0; i < 40; ++i) (void)env.tick(1.0f);   // sustain 0.5
  for (int i = 0; i < 10; ++i) (void)env.tick(0.0f);   // partway released
  const float floor_level = env.tick(0.0f);
  CHECK(floor_level > 0.0f);
  const float retrig = env.tick(1.0f);  // rising edge again
  CHECK(retrig >= floor_level);          // no dip toward zero
  float peak = retrig;
  for (int i = 0; i < 12; ++i) peak = std::max(peak, env.tick(1.0f));
  CHECK(peak == doctest::Approx(1.0f));  // completes the new attack
}

TEST_CASE("curves shape the attack without changing its duration") {
  AdsrRig linear("lin");
  AdsrRig exponential("exp");
  AdsrRig logarithmic("log");
  float lin_mid = 0.0f, exp_mid = 0.0f, log_mid = 0.0f;
  for (int i = 0; i < 5; ++i) {
    lin_mid = linear.tick(1.0f);
    exp_mid = exponential.tick(1.0f);
    log_mid = logarithmic.tick(1.0f);
  }
  CHECK(exp_mid < lin_mid);
  CHECK(log_mid > lin_mid);
  for (int i = 5; i < 10; ++i) {
    lin_mid = linear.tick(1.0f);
    exp_mid = exponential.tick(1.0f);
    log_mid = logarithmic.tick(1.0f);
  }
  CHECK(lin_mid == doctest::Approx(1.0f));
  CHECK(exp_mid == doctest::Approx(1.0f));
  CHECK(log_mid == doctest::Approx(1.0f));
}

TEST_CASE("idle envelope stays at zero and output range is unipolar") {
  AdsrRig env;
  for (int i = 0; i < 100; ++i) CHECK(env.tick(0.0f) == doctest::Approx(0.0f));

  for (int i = 0; i < 60; ++i) {
    const float o = env.tick(1.0f);
    CHECK(o >= 0.0f);
    CHECK(o <= 1.0f + 1e-6f);  // sustain clamps to 1
  }
}
```

```cpp
#include <algorithm>
#include <cmath>
#include "dsp/modules.h"

namespace ms::dsp {
namespace {

constexpr InputInfo kIns[] = {{"gate", 0.f},
                              {"attack", 0.005f},
                              {"decay", 0.15f},
                              {"sustain", 0.7f},
                              {"release", 0.3f}};
constexpr const char* kCurves[] = {"lin", "exp", "log", nullptr};
constexpr ConfigInfo kCfg[] = {{"curve", kCurves}};
constexpr OutputInfo kOuts[] = {{"out"}};

class Adsr final : public Module {
 public:
  void init(double rate, const Config& cfg) override {
    rate_ = rate;
    const std::string curve = cfg_str(cfg, "curve", "lin");
    curve_ = curve == "exp" ? 1 : curve == "log" ? 2 : 0;
    stage_ = Stage::Idle;
    level_ = 0.0;
    start_ = 0.0;
    progress_ = 0.0;
    prev_gate_ = 0.0f;
  }

  void process(const float* ins, float* outs) override {
    const bool high = ins[0] > 0.5f;
    const bool triggered = high && prev_gate_ <= 0.5f;
    prev_gate_ = ins[0];

    if (triggered) {
      start_ = level_;
      progress_ = 0.0;
      stage_ = Stage::Attack;
    } else if (!high && stage_ != Stage::Idle && stage_ != Stage::Release) {
      start_ = level_;
      progress_ = 0.0;
      stage_ = Stage::Release;
    }

    const double sus = std::clamp(static_cast<double>(ins[3]), 0.0, 1.0);
    switch (stage_) {
      case Stage::Attack:
        progress_ = std::min(1.0, progress_ + increment(ins[1]));
        level_ = start_ + (1.0 - start_) * shape(progress_);
        if (progress_ >= 1.0) {
          level_ = 1.0;
          start_ = 1.0;
          progress_ = 0.0;
          stage_ = Stage::Decay;
        }
        break;
      case Stage::Decay:
        progress_ = std::min(1.0, progress_ + increment(ins[2]));
        // Live sustain: the tail bends toward wherever sustain is now.
        level_ = sus + (1.0 - sus) * (1.0 - shape(progress_));
        if (progress_ >= 1.0) {
          level_ = sus;
          stage_ = Stage::Sustain;
        }
        break;
      case Stage::Sustain:
        level_ = sus;
        break;
      case Stage::Release:
        progress_ = std::min(1.0, progress_ + increment(ins[4]));
        level_ = start_ * (1.0 - shape(progress_));
        if (progress_ >= 1.0) {
          level_ = 0.0;
          stage_ = Stage::Idle;
        }
        break;
      case Stage::Idle:
        level_ = 0.0;
        break;
    }

    outs[0] = static_cast<float>(level_);
  }

  void copy_state_from(const Module& other) override {
    const auto& o = static_cast<const Adsr&>(other);
    stage_ = o.stage_;
    level_ = o.level_;
    start_ = o.start_;
    progress_ = o.progress_;
    prev_gate_ = o.prev_gate_;
  }

 private:
  enum class Stage { Idle, Attack, Decay, Sustain, Release };

  // Same contract as func: double accumulation — float increments stall past
  // 2^25 samples and the envelope would freeze short of its target.
  double increment(float seconds) const {
    const double samples = std::max(seconds, 0.0f) * rate_;
    return samples <= 1.0 ? 1.0 : 1.0 / samples;
  }

  double shape(double x) const {
    if (curve_ == 1) return x * x;
    if (curve_ == 2) return std::sqrt(x);
    return x;
  }

  double rate_ = 48000.0;
  int curve_ = 0;
  Stage stage_ = Stage::Idle;
  double level_ = 0.0;
  double start_ = 0.0;
  double progress_ = 0.0;
  float prev_gate_ = 0.0f;
};

std::unique_ptr<Module> make() { return std::make_unique<Adsr>(); }
}  // namespace

const ModuleInfo adsr_info{"adsr", kIns, kCfg, kOuts, &make};
}  // namespace ms::dsp
```

```cpp
TEST_CASE("adsr migrates an active decay with the gate held high") {
  const ModuleInfo* info = find_module("adsr");
  REQUIRE(info != nullptr);
  auto a = info->create();
  auto b = info->create();
  a->init(1000.0, {});
  b->init(1000.0, {});
  // gate, attack, decay, sustain, release
  float ins[5] = {1.0f, 0.01f, 0.2f, 0.4f, 0.05f};
  float out_a[1], out_b[1];
  for (int i = 0; i < 60; ++i) a->process(ins, out_a);  // mid-decay
  b->copy_state_from(*a);
  // Gate stays high across the copy: a fresh prev_gate_ (0.0) would see a
  // spurious rising edge and restart the attack, diverging immediately.
  for (int i = 0; i < 80; ++i) {
    a->process(ins, out_a);
    b->process(ins, out_b);
    CHECK(out_a[0] == doctest::Approx(out_b[0]));
  }
}

TEST_CASE("adsr migrates a release tail without re-idling early") {
  const ModuleInfo* info = find_module("adsr");
  REQUIRE(info != nullptr);
  auto a = info->create();
  auto b = info->create();
  a->init(1000.0, {});
  b->init(1000.0, {});
  float ins[5] = {1.0f, 0.01f, 0.02f, 0.6f, 0.5f};
  float out_a[1], out_b[1];
  for (int i = 0; i < 50; ++i) a->process(ins, out_a);  // through to sustain
  ins[0] = 0.0f;
  for (int i = 0; i < 100; ++i) a->process(ins, out_a);  // 100/500 through release
  b->copy_state_from(*a);
  // A dropped start_ or progress_ copy makes b's tail restart from the wrong
  // height or the wrong point on the curve.
  for (int i = 0; i < 120; ++i) {
    a->process(ins, out_a);
    b->process(ins, out_b);
    CHECK(out_a[0] == doctest::Approx(out_b[0]));
  }
}
```

```
# keys -- classic subtractive keys: two detuned saws, ladder filter, and TWO
# adsr envelopes (amp + filter) gated by the clock like a held keyboard.
# curve: log gives the fast-then-slow analog fall. The clock gate is high for
# half of each beat, so the sustain stage is audible before every release.

clk = clock { bpm: 100 }

# ---- melody: wandering voltage snapped to A natural minor ----
wander = random { rate: 1.3, smooth: 0, seed: 21 }
notes = sh { mode: sample }
wander -> notes.in
clk -> notes.trig
scale = quant { scale: [0, 2, 3, 5, 7, 8, 10] }
notes -> scale.in : 0.35

# ---- voice: two saws detuned by ~7 cents ----
osc1 = vco { freq: 220hz, wave: saw }
osc2 = vco { freq: 220hz, wave: saw, v_oct: 0.006 }
scale -> osc1.v_oct
scale -> osc2.v_oct

# ---- the point of this patch: amp ADSR and filter ADSR from one gate ----
amp_env = adsr { attack: 0.004, decay: 0.18, sustain: 0.55, release: 0.25, curve: log }
clk -> amp_env.gate
filt_env = adsr { attack: 0.002, decay: 0.22, sustain: 0.2, release: 0.2, curve: log }
clk -> filt_env.gate

vcf = ladder { cutoff: 220, res: 0.45 }
osc1 -> vcf.in : 0.5
osc2 -> vcf.in : 0.5
filt_env -> vcf.cutoff : 3200

amp = vca { gain: 0, response: exp }
vcf -> amp.in
amp_env -> amp.cv : 0.9

# ---- space: echo hand-tuned to the tempo (dotted eighth at 100 BPM) ----
echo = delay { time: 450ms, feedback: 0.4, tone: 3000, mix: 0.3 }
amp -> echo.in

echo -> out.in : 0.85
-0.15 -> out.level
```
