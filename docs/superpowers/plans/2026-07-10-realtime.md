# modsynth Realtime + Hot-Reload (Phase 1b) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `modsynth play patches/first.patch` — live PipeWire audio that hot-reloads on save: param edits glide over 20 ms, topology edits crossfade over 50 ms with per-module state migrated by (name,type), bad patches keep the last good graph playing, and the RT thread never allocates, locks, or does I/O.

**Architecture:** A lock-free `SpscQueue` carries commands into the RT thread (param glides, engine swaps, watch taps) and retired engines back out for non-RT deletion. `RtRunner` is the RT-side state machine (current/fading engine, crossfade, meters); `Controller` is its non-RT companion (parse → compile → diff against a shadow graph → emit commands). `PipewireOut` wraps a `pw_filter` with two float32 DSP ports; `PatchWatcher` wraps inotify-on-parent-directory with a 50 ms debounce. Everything except `PipewireOut` is deterministic and tested offline; `RtRunner` + `Controller` are exercised single-threaded in tests plus a threaded stress under ASan/UBSan.

**Tech Stack:** C++20 · CMake/CMakePresets · doctest · PipeWire (`libpipewire-0.3`, verified 1.6.7 installed) · inotify · mise tasks.

`src/rt/spsc.h`:
```cpp
#pragma once
#include <array>
#include <atomic>
#include <cstddef>
#include <type_traits>

namespace ms::rt {

// Lock-free single-producer single-consumer ring. Exactly one thread may call
// try_push and exactly one thread may call try_pop. Wait-free, no allocation.
// Usable capacity is Cap-1: `next == head` disambiguates full from empty.
// Memory ordering: the payload write must be visible before the index that
// publishes it, so each index is stored with release and read by the peer
// with acquire; each side reads its own index relaxed (nobody else writes it).
// Cached peer indices avoid touching the other core's cache line on the
// common path (moodycamel optimization); alignas(64) prevents false sharing.
// See docs/research/2026-07-10-rt-spsc-discipline.md.
template <typename T, size_t Cap>
class SpscQueue {
  static_assert(std::is_trivially_copyable_v<T>);
  static_assert(Cap >= 2 && (Cap & (Cap - 1)) == 0, "Cap must be a power of 2");

 public:
  bool try_push(const T& v) {
    const size_t t = tail_.load(std::memory_order_relaxed);
    const size_t next = (t + 1) & kMask;
    if (next == cached_head_) {
      cached_head_ = head_.load(std::memory_order_acquire);
      if (next == cached_head_) return false;  // truly full
    }
    buf_[t] = v;
    tail_.store(next, std::memory_order_release);
    return true;
  }

  bool try_pop(T& out) {
    const size_t h = head_.load(std::memory_order_relaxed);
    if (h == cached_tail_) {
      cached_tail_ = tail_.load(std::memory_order_acquire);
      if (h == cached_tail_) return false;  // truly empty
    }
    out = buf_[h];
    head_.store((h + 1) & kMask, std::memory_order_release);
    return true;
  }

 private:
  static constexpr size_t kMask = Cap - 1;
  alignas(64) std::atomic<size_t> head_{0};  // consumer advances, producer reads
  alignas(64) std::atomic<size_t> tail_{0};  // producer advances, consumer reads
  alignas(64) size_t cached_head_ = 0;       // producer-local
  alignas(64) size_t cached_tail_ = 0;       // consumer-local
  alignas(64) std::array<T, Cap> buf_{};
};

}  // namespace ms::rt
```

`src/rt/denormal.h`:
```cpp
#pragma once

#if defined(__SSE2__)
#include <pmmintrin.h>
#include <xmmintrin.h>
#endif

namespace ms::rt {

// Denormal floats cost 10-100x on x86; feedback tails decay into them.
// FTZ/DAZ live in the per-thread MXCSR, so this must run on the thread that
// does the math -- call it at the top of every audio callback (cheap), not
// once at startup: we don't own the RT thread's creation.
inline void flush_denormals_on_this_thread() {
#if defined(__SSE2__)
  _MM_SET_FLUSH_ZERO_MODE(_MM_FLUSH_ZERO_ON);
  _MM_SET_DENORMAL_ZERO_MODE(_MM_DENORMALS_ZERO_ON);
#endif
}

}  // namespace ms::rt
```

```cpp
class Engine {

  void process(float* left, float* right, size_t frames);
  double sample_rate() const { return g_->sample_rate; }

    size_t node = 0, slot = 0;
    float target = 0.0f, step = 0.0f;
    size_t remaining = 0;

  std::vector<size_t> out_offset_; // per node
  std::vector<float> ins_;         // scratch, sized to max input count
```

```cpp
void Engine::set_param(size_t node, size_t slot, float target, size_t glide_samples) {
  if (node >= g_->nodes.size()) return;
  auto& base = g_->nodes[node].input_base;
  if (slot >= base.size()) return;
  for (size_t i = 0; i < n_glides_; ++i)
    if (glides_[i].node == node && glides_[i].slot == slot) {
      glides_[i] = glides_[--n_glides_];  // replace in-flight glide on this param
      break;
    }
  if (glide_samples == 0 || n_glides_ == kMaxGlides) {
    base[slot] = target;
    return;
  }
  glides_[n_glides_++] = Glide{node, slot, target,
      (target - base[slot]) / static_cast<float>(glide_samples), glide_samples};
}

void Engine::step_glides() {
  for (size_t i = 0; i < n_glides_;) {
    Glide& gl = glides_[i];
    float& base = g_->nodes[gl.node].input_base[gl.slot];
    base += gl.step;
    if (--gl.remaining == 0) {
      base = gl.target;                 // land exactly, no float drift
      glides_[i] = glides_[--n_glides_];
    } else {
      ++i;
    }
  }
}

float Engine::output_value(size_t node, size_t slot) const {
  if (node >= g_->nodes.size()) return 0.0f;
  if (slot >= g_->nodes[node].info->outputs.size()) return 0.0f;
  return outs_[out_offset_[node] + slot];
}
```

```cpp
namespace ms::graph {
namespace {

bool same_config_value(const dsp::ConfigValue& a, const dsp::ConfigValue& b) {
  return a.is_num == b.is_num && a.num == b.num && a.str == b.str && a.arr == b.arr;
}

bool same_config(const dsp::Config& a, const dsp::Config& b) {
  if (a.size() != b.size()) return false;
  for (size_t i = 0; i < a.size(); ++i)
    if (a[i].first != b[i].first || !same_config_value(a[i].second, b[i].second))
      return false;
  return true;
}

bool same_cables(const std::vector<Cable>& a, const std::vector<Cable>& b) {
  if (a.size() != b.size()) return false;
  for (size_t i = 0; i < a.size(); ++i)
    if (a[i].src_mod != b[i].src_mod || a[i].src_out != b[i].src_out ||
        a[i].dst_mod != b[i].dst_mod || a[i].dst_in != b[i].dst_in ||
        a[i].depth != b[i].depth)
      return false;
  return true;
}

// Identical structure: same nodes (name, type, config) in the same order and
// the same cables. Compile is deterministic, so a param-only source edit
// reproduces the old ordering exactly.
bool same_structure(const Graph& a, const Graph& b) {
  if (a.nodes.size() != b.nodes.size()) return false;
  for (size_t i = 0; i < a.nodes.size(); ++i) {
    if (a.nodes[i].name != b.nodes[i].name) return false;
    if (a.nodes[i].info != b.nodes[i].info) return false;
    if (!same_config(a.nodes[i].config, b.nodes[i].config)) return false;
  }
  if (a.in_cables.size() != b.in_cables.size()) return false;
  for (size_t i = 0; i < a.in_cables.size(); ++i)
    if (!same_cables(a.in_cables[i], b.in_cables[i])) return false;
  return true;
}

}  // namespace

ReloadPlan plan_reload(const Graph& old_g, Graph& new_g) {
  ReloadPlan plan;
  if (same_structure(old_g, new_g)) {
    for (size_t n = 0; n < new_g.nodes.size(); ++n)
      for (size_t s = 0; s < new_g.nodes[n].input_base.size(); ++s)
        if (new_g.nodes[n].input_base[s] != old_g.nodes[n].input_base[s])
          plan.params.push_back({static_cast<int>(n), static_cast<int>(s),
                                 new_g.nodes[n].input_base[s]});
    return plan;
  }
  plan.topology = true;
  new_g.migrate_pairs.clear();
  for (size_t ni = 0; ni < new_g.nodes.size(); ++ni)
    for (size_t oi = 0; oi < old_g.nodes.size(); ++oi)
      if (new_g.nodes[ni].name == old_g.nodes[oi].name &&
          new_g.nodes[ni].info == old_g.nodes[oi].info) {
        new_g.migrate_pairs.emplace_back(static_cast<int>(oi), static_cast<int>(ni));
        break;
      }
  return plan;
}

}  // namespace ms::graph
```

**Files:**
- Create: `src/engine/rt_runner.h`, `src/engine/rt_runner.cpp`
- Test: `tests/test_runner.cpp`

**Interfaces:**
- Consumes: `ms::rt::SpscQueue` (Task 1), `Engine::set_param/migrate_from/output_value` (Tasks 3-4).
- Produces (in `ms::engine`):
  ```cpp
  struct Command {  // trivially copyable; flows non-RT -> RT via SPSC
    enum class Kind : uint32_t { SetParam, Swap, Watch };
    Kind kind = Kind::SetParam;
    uint32_t node = 0, slot = 0;   // SetParam/Watch target (Watch: node==kWatchOff disables)
    float value = 0.0f;            // SetParam target
    Engine* engine = nullptr;      // Swap: ownership passes to the runner
  };

  class RtRunner {
   public:
    static constexpr size_t kMaxBlock = 8192;
    static constexpr uint32_t kWatchOff = 0xffffffffu;
    explicit RtRunner(double sample_rate);
    ~RtRunner();  // deletes engines it still owns; only call after RT rendering stopped
    // non-RT side:
    bool push(const Command& c);          // false = queue full (caller retries)
    bool try_retire(Engine*& e);          // drain engines to delete off-RT
    float peak_l() const;                 // decaying peak meters (atomic)
    float peak_r() const;
    bool try_watch_value(float& v);       // drain watch tap samples
    // RT side (the ONLY method the audio thread calls):
    void render(float* l, float* r, size_t frames);
   private:
    // current_: playing engine. old_: previous engine during the 50ms
    // crossfade; retired to the queue when the fade completes.
  };
  ```
  Behavior contract:
  - `render` drains all pending commands at block start, then renders. No engine yet → silence.
  - `Swap` with an engine already playing: `new->migrate_from(*current_)`, old becomes the fade-out side, equal-power crossfade over 50 ms (`gain_new = sin(t·π/2)`, `gain_old = cos(t·π/2)`). A `Swap` arriving mid-fade retires the fading engine immediately (its cos-gain contribution stops; two saves within 50 ms is the rare case and a small step on the *outgoing* graph is acceptable) and starts a fresh fade from the current one.
  - `SetParam` → `current_->set_param(node, slot, value, 20ms_in_samples)`.
  - Meters: per-block peak with linear release of 3.0 full-scale units/second, published via `std::atomic<float>` store(release)/load(acquire) — single writer (RT), single reader.
  - Watch: one `output_value` sample per block pushed to a float SPSC ring (drop-newest when full is fine for a display).
  - `render` with frames > kMaxBlock processes in kMaxBlock chunks (scratch buffers are fixed arrays).
  - EVERY code path in `render` obeys the RT rules: no allocation, no locks, no I/O. `delete` never happens here — engines only leave via the retire queue.

```cpp
#pragma once
#include <array>
#include <atomic>
#include <cstdint>
#include <type_traits>
#include "engine/engine.h"
#include "rt/spsc.h"

namespace ms::engine {

struct Command {
  enum class Kind : uint32_t { SetParam, Swap, Watch };
  Kind kind = Kind::SetParam;
  uint32_t node = 0, slot = 0;
  float value = 0.0f;
  Engine* engine = nullptr;  // Swap: ownership passes to the runner
};
static_assert(std::is_trivially_copyable_v<Command>);

// RT-side state machine. render() is the only method the audio thread calls;
// everything it does is allocation-, lock-, and I/O-free. Engines enter via
// Swap commands and leave via the retire queue (deleted by the non-RT side).
class RtRunner {
 public:
  static constexpr size_t kMaxBlock = 8192;
  static constexpr uint32_t kWatchOff = 0xffffffffu;

  explicit RtRunner(double sample_rate);
  ~RtRunner();
  RtRunner(const RtRunner&) = delete;
  RtRunner& operator=(const RtRunner&) = delete;

  // --- non-RT side ---
  bool push(const Command& c) { return commands_.try_push(c); }
  bool try_retire(Engine*& e) { return retired_.try_pop(e); }
  float peak_l() const { return peak_l_.load(std::memory_order_acquire); }
  float peak_r() const { return peak_r_.load(std::memory_order_acquire); }
  bool try_watch_value(float& v) { return watch_.try_pop(v); }

  // --- RT side ---
  void render(float* l, float* r, size_t frames);

 private:
  void drain_commands();
  void render_block(float* l, float* r, size_t frames);

  double rate_;
  size_t glide_samples_;  // 20ms
  size_t fade_samples_;   // 50ms
  rt::SpscQueue<Command, 1024> commands_;
  rt::SpscQueue<Engine*, 8> retired_;
  rt::SpscQueue<float, 4096> watch_;
  Engine* current_ = nullptr;
  Engine* old_ = nullptr;  // non-null while crossfading
  size_t fade_pos_ = 0;
  uint32_t watch_node_ = kWatchOff, watch_slot_ = 0;
  std::atomic<float> peak_l_{0.0f}, peak_r_{0.0f};
  float env_l_ = 0.0f, env_r_ = 0.0f;
  std::array<float, kMaxBlock> scratch_l_{}, scratch_r_{};
};
```

```cpp
#include "engine/rt_runner.h"
#include <algorithm>
#include <cmath>
#include <numbers>

namespace ms::engine {

RtRunner::RtRunner(double sample_rate)
    : rate_(sample_rate),
      glide_samples_(static_cast<size_t>(sample_rate * 0.020)),
      fade_samples_(static_cast<size_t>(sample_rate * 0.050)) {}

RtRunner::~RtRunner() {
  // Only safe once the audio thread has stopped calling render().
  Command c;
  while (commands_.try_pop(c))
    if (c.kind == Command::Kind::Swap) delete c.engine;
  Engine* e = nullptr;
  while (retired_.try_pop(e)) delete e;
  delete current_;
  delete old_;
}

void RtRunner::drain_commands() {
  Command c;
  while (commands_.try_pop(c)) {
    switch (c.kind) {
      case Command::Kind::SetParam:
        if (current_) current_->set_param(c.node, c.slot, c.value, glide_samples_);
        break;
      case Command::Kind::Swap:
        if (old_) {  // a second save landed mid-fade: retire the fading engine
          retired_.try_push(old_);  // capacity is gated by the controller
          old_ = nullptr;
        }
        if (!current_) {
          current_ = c.engine;  // first graph: adopt immediately
        } else {
          c.engine->migrate_from(*current_);
          old_ = current_;
          current_ = c.engine;
          fade_pos_ = 0;
        }
        break;
      case Command::Kind::Watch:
        watch_node_ = c.node;
        watch_slot_ = c.slot;
        break;
    }
  }
}

void RtRunner::render(float* l, float* r, size_t frames) {
  while (frames > kMaxBlock) {
    render_block(l, r, kMaxBlock);
    l += kMaxBlock;
    r += kMaxBlock;
    frames -= kMaxBlock;
  }
  render_block(l, r, frames);
}

void RtRunner::render_block(float* l, float* r, size_t frames) {
  drain_commands();
  if (!current_) {
    std::fill(l, l + frames, 0.0f);
    std::fill(r, r + frames, 0.0f);
    peak_l_.store(0.0f, std::memory_order_release);
    peak_r_.store(0.0f, std::memory_order_release);
    return;
  }

  current_->process(l, r, frames);

  if (old_) {
    old_->process(scratch_l_.data(), scratch_r_.data(), frames);
    constexpr float kHalfPi = std::numbers::pi_v<float> / 2.0f;
    for (size_t i = 0; i < frames; ++i) {
      const size_t p = fade_pos_ + i;
      const float t = p >= fade_samples_
          ? 1.0f
          : static_cast<float>(p) / static_cast<float>(fade_samples_);
      const float gn = std::sin(t * kHalfPi);
      const float go = std::cos(t * kHalfPi);
      l[i] = l[i] * gn + scratch_l_[i] * go;
      r[i] = r[i] * gn + scratch_r_[i] * go;
    }
    fade_pos_ += frames;
    if (fade_pos_ >= fade_samples_) {
      retired_.try_push(old_);
      old_ = nullptr;
    }
  }

  float pk_l = 0.0f, pk_r = 0.0f;
  for (size_t i = 0; i < frames; ++i) {
    pk_l = std::max(pk_l, std::abs(l[i]));
    pk_r = std::max(pk_r, std::abs(r[i]));
  }
  const float release = 3.0f * static_cast<float>(frames) / static_cast<float>(rate_);
  env_l_ = std::max(pk_l, env_l_ - release);
  env_r_ = std::max(pk_r, env_r_ - release);
  peak_l_.store(env_l_, std::memory_order_release);
  peak_r_.store(env_r_, std::memory_order_release);

  if (watch_node_ != kWatchOff)
    watch_.try_push(current_->output_value(watch_node_, watch_slot_));
}

}  // namespace ms::engine
```

  ```cpp
   public:
    Controller(RtRunner& runner, double sample_rate);
    struct LoadResult {
      bool ok = false;

      Diags errors;            // parse/compile diagnostics when !ok
  ```

  Design notes the implementer must respect:
  - The controller keeps a **shadow graph**: a second compile of the last good
    source that never leaves this thread. All diffing happens against the
    shadow, never against the graph the RT thread is playing — that is what
    makes reload planning race-free. After a param-only load the shadow is
    replaced by the new compile (same structure, fresh bases); after a
    topology load the source is compiled **twice** (one graph → Engine → RT,
    one → shadow).
  - Backpressure gate: a topology load is refused with a "reload still in
    flight" Diag if more than 2 engines are outstanding (current + one
    fading). This keeps the retire queue provably unfillable. `load` calls
    `drain_retired()` first, so in practice this only triggers under a
    save-flood faster than the 50 ms fade.
  - Param commands are pushed one by one; if the command queue is full
    (pathological), remaining params are dropped and the load still reports
    ok — the next save re-syncs them. Watch resolution: default port = output
    slot 0; a named port uses `ModuleInfo::output_index`.

`src/engine/controller.cpp`:
```cpp
#include "engine/controller.h"
#include "dsl/parser.h"
#include "graph/reload.h"

namespace ms::engine {

Controller::Controller(RtRunner& runner, double sample_rate)
    : runner_(runner), rate_(sample_rate) {}

Controller::LoadResult Controller::load(std::string_view src) {
  LoadResult res;
  drain_retired();

  auto p = dsl::parse(src);
  if (!p.ok()) {
    res.errors = std::move(p.errors);
    return res;
  }
  auto c = graph::compile(p.ast, rate_);
  if (!c.ok()) {
    res.errors = std::move(c.errors);
    return res;
  }

  if (shadow_) {
    auto plan = graph::plan_reload(*shadow_, *c.graph);
    if (!plan.topology) {
      for (const auto& pc : plan.params) {
        Command cmd;
        cmd.kind = Command::Kind::SetParam;
        cmd.node = static_cast<uint32_t>(pc.node);
        cmd.slot = static_cast<uint32_t>(pc.slot);
        cmd.value = pc.target;
        if (!runner_.push(cmd)) break;  // full queue: next save re-syncs
      }
      shadow_ = std::move(c.graph);  // same structure, fresh bases
      res.ok = true;
      return res;
    }
  }

  // Topology path (or first load). Gate: keep the retire queue unfillable.
  if (engines_in_flight() >= 2) {
    res.errors.push_back({0, 0, "reload still in flight; save again in a moment"});
    return res;
  }

  // The RT engine and the shadow must be distinct objects: compile again.
  auto c2 = graph::compile(p.ast, rate_);
  Command cmd;
  cmd.kind = Command::Kind::Swap;
  cmd.engine = new Engine(std::move(c.graph));
  if (!runner_.push(cmd)) {
    delete cmd.engine;
    res.errors.push_back({0, 0, "engine command queue full; try again"});
    return res;
  }
  ++pushed_;
  shadow_ = std::move(c2.graph);
  res.ok = true;
  res.topology = true;
  if (!watch_spec_.empty()) push_watch_for_shadow();
  return res;
}

bool Controller::set_watch(std::string_view spec) {
  std::string keep = watch_spec_;
  watch_spec_ = std::string(spec);
  if (push_watch_for_shadow()) return true;
  watch_spec_ = std::move(keep);
  return false;
}

void Controller::clear_watch() {
  watch_spec_.clear();
  Command cmd;
  cmd.kind = Command::Kind::Watch;
  cmd.node = RtRunner::kWatchOff;
  runner_.push(cmd);
}

bool Controller::push_watch_for_shadow() {
  if (!shadow_ || watch_spec_.empty()) return false;
  std::string_view spec = watch_spec_;
  std::string_view mod = spec, port;
  if (auto dot = spec.find('.'); dot != std::string_view::npos) {
    mod = spec.substr(0, dot);
    port = spec.substr(dot + 1);
  }
  for (size_t i = 0; i < shadow_->nodes.size(); ++i) {
    const auto& n = shadow_->nodes[i];
    if (n.name != mod) continue;
    int slot = 0;
    if (!port.empty()) {
      slot = n.info->output_index(port);
      if (slot < 0) return false;
    }
    Command cmd;
    cmd.kind = Command::Kind::Watch;
    cmd.node = static_cast<uint32_t>(i);
    cmd.slot = static_cast<uint32_t>(slot);
    return runner_.push(cmd);
  }
  return false;
}

void Controller::drain_retired() {
  Engine* e = nullptr;
  while (runner_.try_retire(e)) {
    delete e;
    ++reaped_;
  }
}

}  // namespace ms::engine
```

```cpp
#pragma once
#include <chrono>
#include <string>

namespace ms::io {

// Watches one patch file for saves, robust against editor temp-file+rename
// saves: watches the parent directory and matches the basename. Debounces
// with a 50ms quiescence window (see docs/research hotreload brief).
class PatchWatcher {
 public:
  explicit PatchWatcher(const std::string& patch_path);
  ~PatchWatcher();
  PatchWatcher(const PatchWatcher&) = delete;
  PatchWatcher& operator=(const PatchWatcher&) = delete;

  bool ok() const { return fd_ >= 0 && wd_ >= 0; }
  bool poll_changed(int timeout_ms);

 private:
  void arm();

  int fd_ = -1;
  int wd_ = -1;
  std::string dir_, base_;
  bool pending_ = false;
  std::chrono::steady_clock::time_point deadline_{};
};

}  // namespace ms::io
```

`src/io/watcher.cpp`:
```cpp
#include "io/watcher.h"

#include <poll.h>
#include <sys/inotify.h>
#include <unistd.h>

#include <climits>
#include <cstring>
#include <filesystem>

namespace ms::io {

namespace {
constexpr auto kDebounce = std::chrono::milliseconds(50);
constexpr uint32_t kDirMask =
    IN_CLOSE_WRITE | IN_MOVED_TO | IN_MOVE_SELF | IN_DELETE_SELF;
}  // namespace

PatchWatcher::PatchWatcher(const std::string& patch_path) {
  std::filesystem::path p(patch_path);
  dir_ = p.has_parent_path() ? p.parent_path().string() : ".";
  base_ = p.filename().string();
  fd_ = inotify_init1(IN_NONBLOCK | IN_CLOEXEC);
  if (fd_ >= 0) arm();
}

void PatchWatcher::arm() {
  wd_ = inotify_add_watch(fd_, dir_.c_str(), kDirMask);
}

PatchWatcher::~PatchWatcher() {
  if (fd_ >= 0) close(fd_);
}

bool PatchWatcher::poll_changed(int timeout_ms) {
  if (!ok()) return false;
  const auto now = std::chrono::steady_clock::now();

  int wait_ms = timeout_ms;
  if (pending_) {
    const auto until = std::chrono::duration_cast<std::chrono::milliseconds>(
        deadline_ - now).count();
    if (until <= 0) {  // quiescence reached
      pending_ = false;
      return true;
    }
    wait_ms = std::min<long long>(wait_ms, until);
  }

  pollfd pfd{fd_, POLLIN, 0};
  if (poll(&pfd, 1, wait_ms) > 0 && (pfd.revents & POLLIN)) {
    // NOLINTNEXTLINE(cppcoreguidelines-avoid-c-arrays): kernel ABI buffer
    alignas(inotify_event) char buf[sizeof(inotify_event) + NAME_MAX + 1];
    ssize_t n = 0;
    while ((n = read(fd_, buf, sizeof(buf))) > 0) {
      for (char* ptr = buf; ptr < buf + n;) {
        const auto* ev = reinterpret_cast<const inotify_event*>(ptr);
        if (ev->mask & (IN_MOVE_SELF | IN_DELETE_SELF)) {
          arm();  // directory itself replaced: re-watch
        } else if ((ev->mask & (IN_CLOSE_WRITE | IN_MOVED_TO)) && ev->len > 0 &&
                   base_ == ev->name) {
          pending_ = true;
          deadline_ = std::chrono::steady_clock::now() + kDebounce;
        }
        ptr += sizeof(inotify_event) + ev->len;
      }
    }
  }

  if (pending_ && std::chrono::steady_clock::now() >= deadline_) {
    pending_ = false;
    return true;
  }
  return false;
}

}  // namespace ms::io
```

  ```cpp
  class PipewireOut {
   public:
    // fn is called on PipeWire's RT data thread. Everything it does must be
    // RT-safe. `user` is opaque. latency is a PW_KEY_NODE_LATENCY string like
    // "256/48000", or nullptr for the server default.
    using RenderFn = void (*)(void* user, float* l, float* r, size_t frames);
    PipewireOut(const char* app_name, RenderFn fn, void* user,
                const char* latency, uint32_t requested_rate);
    ~PipewireOut();
    PipewireOut(const PipewireOut&) = delete;
    PipewireOut& operator=(const PipewireOut&) = delete;
    bool ok() const;                 // connected without error so far
    std::string error() const;       // last state_changed error ("" if none)
    uint32_t observed_rate() const;  // clock rate seen in process (0 = none yet)
    uint64_t xruns() const;          // cycles flagged SPA_IO_CLOCK_FLAG_XRUN_RECOVER
  };
  ```
  **Read `docs/research/2026-07-10-pipewire-filter-api.md` first** — it was
  verified against the exact headers installed here (libpipewire-0.3 1.6.7)
  and contains a compile-tested skeleton. Key requirements it explains:
  - `pw_thread_loop` (CLI thread stays free); every non-loop-thread `pw_*`
    call inside `pw_thread_loop_lock`/`unlock`; `process` is the exception.
  - Two OUTPUT DSP ports (`PW_KEY_FORMAT_DSP` "32 bit float mono audio",
    channels FL/FR), `PW_FILTER_FLAG_RT_PROCESS`, media class
    `Stream/Output/Audio` + autoconnect (the shipped example's
    `MEDIA_CATEGORY "Source"` is wrong for a playback stream — use
    "Playback").
  - `pw_filter_get_dsp_buffer` **can return NULL** — skip the cycle.
  - `n_samples = position->clock.duration` varies per cycle; rate is
    `position->clock.rate.denom`, re-read every call.
  - Teardown order: `pw_thread_loop_stop` (WITHOUT the lock) →
    `pw_filter_destroy` → `pw_thread_loop_destroy` → `pw_deinit`.
  - PipeWire headers trip `-Wconversion`: include them ONLY inside the
    documented `#pragma GCC diagnostic` wrapper, confined to
    `pipewire_out.cpp`. The header `pipewire_out.h` must not include any
    PipeWire header (pimpl via `struct Impl`).
  - In the process callback, call `rt::flush_denormals_on_this_thread()`
    first (per-thread MXCSR; we don't own the thread's creation).

There are no unit tests for this task (tests must stay hermetic — no PipeWire
server dependency). The gates are: `mise run ci` builds warning-clean, all

`src/engine/pipewire_out.cpp`:
```cpp
#include "engine/pipewire_out.h"

#pragma GCC diagnostic push
#pragma GCC diagnostic ignored "-Wconversion"
#pragma GCC diagnostic ignored "-Wsign-conversion"
#include <pipewire/filter.h>
#include <pipewire/pipewire.h>
#include <spa/node/io.h>
#pragma GCC diagnostic pop

#include <array>
#include <atomic>
#include <cstdio>
#include <mutex>

#include "rt/denormal.h"

namespace ms::engine {

struct PipewireOut::Impl {
  RenderFn fn = nullptr;
  void* user = nullptr;
  pw_thread_loop* loop = nullptr;
  pw_filter* filter = nullptr;
  void* port_l = nullptr;  // pw port data handles
  void* port_r = nullptr;
  std::atomic<bool> ok{false};
  std::atomic<uint32_t> rate{0};
  std::atomic<uint64_t> xruns{0};
  std::mutex err_mu;              // non-RT only (state_changed thread)
  std::string err;

  // Fallback silence sink if one dsp buffer is missing this cycle.
  static void on_process(void* userdata, spa_io_position* position) {
    auto* self = static_cast<Impl*>(userdata);
    rt::flush_denormals_on_this_thread();
    const uint32_t n = position->clock.duration;
    self->rate.store(position->clock.rate.denom, std::memory_order_relaxed);
    if (position->clock.flags & SPA_IO_CLOCK_FLAG_XRUN_RECOVER)
      self->xruns.fetch_add(1, std::memory_order_relaxed);
    auto* l = static_cast<float*>(pw_filter_get_dsp_buffer(self->port_l, n));
    auto* r = static_cast<float*>(pw_filter_get_dsp_buffer(self->port_r, n));
    if (l == nullptr || r == nullptr) return;  // port not negotiated this cycle
    self->fn(self->user, l, r, n);
  }

  static void on_state_changed(void* userdata, pw_filter_state,
                               pw_filter_state state, const char* error) {
    auto* self = static_cast<Impl*>(userdata);
    if (state == PW_FILTER_STATE_ERROR) {
      self->ok.store(false, std::memory_order_release);
      std::lock_guard<std::mutex> lk(self->err_mu);
      self->err = error != nullptr ? error : "unknown PipeWire error";
    } else if (state == PW_FILTER_STATE_STREAMING ||
               state == PW_FILTER_STATE_PAUSED) {
      self->ok.store(true, std::memory_order_release);
    }
  }
};

namespace {
const pw_filter_events kEvents = {
    PW_VERSION_FILTER_EVENTS,
    nullptr,                                // destroy
    PipewireOut::Impl::on_state_changed,    // state_changed
    nullptr,                                // io_changed
    nullptr,                                // param_changed
    nullptr,                                // add_buffer
    nullptr,                                // remove_buffer
    PipewireOut::Impl::on_process,          // process
    nullptr,                                // drained
    nullptr,                                // command
};

void* add_port(pw_filter* f, const char* name, const char* channel) {
  return pw_filter_add_port(
      f, PW_DIRECTION_OUTPUT, PW_FILTER_PORT_FLAG_MAP_BUFFERS, 0,
      pw_properties_new(PW_KEY_FORMAT_DSP, "32 bit float mono audio",
                        PW_KEY_PORT_NAME, name,
                        PW_KEY_AUDIO_CHANNEL, channel, nullptr),
      nullptr, 0);
}
}  // namespace

PipewireOut::PipewireOut(const char* app_name, RenderFn fn, void* user,
                         const char* latency, uint32_t requested_rate)
    : impl_(std::make_unique<Impl>()) {
  impl_->fn = fn;
  impl_->user = user;
  pw_init(nullptr, nullptr);
  impl_->loop = pw_thread_loop_new("modsynth-audio", nullptr);
  if (impl_->loop == nullptr) return;

  pw_thread_loop_lock(impl_->loop);
  pw_properties* props = pw_properties_new(
      PW_KEY_MEDIA_TYPE, "Audio",
      PW_KEY_MEDIA_CATEGORY, "Playback",
      PW_KEY_MEDIA_ROLE, "Music",
      PW_KEY_MEDIA_CLASS, "Stream/Output/Audio",
      PW_KEY_NODE_AUTOCONNECT, "true", nullptr);
  if (latency != nullptr)
    pw_properties_set(props, PW_KEY_NODE_LATENCY, latency);
  if (requested_rate != 0) {
    char rate_str[32];
    std::snprintf(rate_str, sizeof(rate_str), "1/%u", requested_rate);
    pw_properties_set(props, PW_KEY_NODE_RATE, rate_str);
  }
  impl_->filter = pw_filter_new_simple(pw_thread_loop_get_loop(impl_->loop),
                                       app_name, props, &kEvents, impl_.get());
  if (impl_->filter != nullptr) {
    impl_->port_l = add_port(impl_->filter, "out_l", "FL");
    impl_->port_r = add_port(impl_->filter, "out_r", "FR");
    if (pw_filter_connect(impl_->filter, PW_FILTER_FLAG_RT_PROCESS, nullptr, 0) < 0) {
      std::lock_guard<std::mutex> lk(impl_->err_mu);
      impl_->err = "pw_filter_connect failed";
    }
  }
  pw_thread_loop_unlock(impl_->loop);
  pw_thread_loop_start(impl_->loop);
}

PipewireOut::~PipewireOut() {
  if (impl_->loop != nullptr) {
    pw_thread_loop_stop(impl_->loop);  // without the lock (documented order)
    if (impl_->filter != nullptr) pw_filter_destroy(impl_->filter);
    pw_thread_loop_destroy(impl_->loop);
  }
  pw_deinit();
}

bool PipewireOut::ok() const {
  return impl_->ok.load(std::memory_order_acquire) && error().empty();
}

std::string PipewireOut::error() const {
  std::lock_guard<std::mutex> lk(impl_->err_mu);
  return impl_->err;
}

uint32_t PipewireOut::observed_rate() const {
  return impl_->rate.load(std::memory_order_relaxed);
}

uint64_t PipewireOut::xruns() const {
  return impl_->xruns.load(std::memory_order_relaxed);
}

}  // namespace ms::engine
```
Implementation notes:
- The `pw_filter_events` designated-initializer list above is ORDER-SENSITIVE
  against the local header (1.6.7): verify the member order in
  `/usr/include/pipewire-0.3/pipewire/filter.h` and prefer C++20 designated
  initializers (`.state_changed = ..., .process = ...`) if the compiler
  accepts them for this struct — that is what the research brief's verified
  skeleton uses.
- `port_l`/`port_r` are pw-owned port-data pointers; we pass size 0 and keep
  no per-port state (the research skeleton used a per-port struct; we don't
  need one — the engine holds all state).
- If the port-data-size-0 variant misbehaves (get_dsp_buffer needs the port
  data pointer as its handle), fall back to `sizeof(void*)` sized port data
  exactly as the research skeleton does — the handle IS the port data pointer
  returned by `pw_filter_add_port`.

```cpp
namespace {

std::atomic<bool> g_stop{false};
void on_sigint(int) { g_stop.store(true); }

void render_trampoline(void* user, float* l, float* r, size_t frames) {
  static_cast<ms::engine::RtRunner*>(user)->render(l, r, frames);
}

void print_status(ms::engine::RtRunner& run,
                  const ms::engine::PipewireOut& pw,
                  bool watching) {
  auto bar = [](float v) {
    std::string b(20, '-');
    int fill = static_cast<int>(std::clamp(v, 0.0f, 1.0f) * 20.0f);
    for (int i = 0; i < fill; ++i) b[static_cast<size_t>(i)] = '#';
    return b;
  };
  std::string line = "\r  L |" + bar(run.peak_l()) + "|  R |" + bar(run.peak_r()) +
                     "|  xruns " + std::to_string(pw.xruns());
  if (watching) {
    float v = 0.0f, last = 0.0f;
    bool got = false;
    while (run.try_watch_value(v)) { last = v; got = true; }
    if (got) {
      char buf[32];
      std::snprintf(buf, sizeof(buf), "  watch=% .3f", static_cast<double>(last));
      line += buf;
    }
  }
  std::fputs(line.c_str(), stderr);
  std::fflush(stderr);
}

int cmd_play(const std::string& patch, const char* latency, double rate,
             const char* watch) {
  std::string src;
  if (!read_file(patch, src)) {
    std::fprintf(stderr, "error: cannot read '%s'\n", patch.c_str());
    return 1;
  }

  ms::engine::RtRunner runner(rate);
  ms::engine::Controller ctl(runner, rate);

  auto report = [&](const ms::Diags& ds) {
    std::fputs("\n", stderr);
    for (const auto& d : ds)
      std::fprintf(stderr, "%s:%d:%d: error: %s\n", patch.c_str(), d.line,
                   d.col, d.msg.c_str());
  };

  auto first = ctl.load(src);
  if (!first.ok) { report(first.errors); return 1; }
  if (watch != nullptr && !ctl.set_watch(watch)) {
    std::fprintf(stderr, "error: cannot watch '%s' (unknown module or port)\n", watch);
    return 1;
  }

  auto pw = std::make_unique<ms::engine::PipewireOut>(
      "modsynth", &render_trampoline, &runner, latency,
      static_cast<uint32_t>(rate));
  ms::io::PatchWatcher watcher(patch);
  if (!watcher.ok())
    std::fprintf(stderr, "warning: cannot watch '%s'; hot-reload disabled\n",
                 patch.c_str());

  std::signal(SIGINT, &on_sigint);
  std::fprintf(stderr, "playing %s  (edit + save to hot-reload, Ctrl-C to quit)\n",
               patch.c_str());

  bool warned_rate = false;
  while (!g_stop.load()) {
    if (watcher.ok() && watcher.poll_changed(100)) {
      if (!read_file(patch, src)) {
        std::fprintf(stderr, "\nerror: cannot re-read '%s'\n", patch.c_str());
      } else {
        auto res = ctl.load(src);
        if (!res.ok) report(res.errors);
        else std::fprintf(stderr, "\nreloaded (%s)\n",
                          res.topology ? "topology" : "params");
      }
    }
    ctl.drain_retired();
    print_status(runner, *pw, watch != nullptr);
    // Spec 7.5: PipeWire disconnects never kill the session. Recreate the
    // backend with a 2s backoff; the graph keeps its state and resumes.
    if (!pw->error().empty()) {
      std::fprintf(stderr, "\nwarning: PipeWire: %s -- reconnecting...\n",
                   pw->error().c_str());
      pw.reset();
      std::this_thread::sleep_for(std::chrono::seconds(2));
      pw = std::make_unique<ms::engine::PipewireOut>(
          "modsynth", &render_trampoline, &runner, latency,
          static_cast<uint32_t>(rate));
    }
    if (!warned_rate && pw->observed_rate() != 0 &&
        pw->observed_rate() != static_cast<uint32_t>(rate)) {
      std::fprintf(stderr,
          "\nwarning: PipeWire runs at %u Hz but the patch was compiled for"
          " %g Hz; restart with --rate %u\n",
          pw->observed_rate(), rate, pw->observed_rate());
      warned_rate = true;
    }
  }
  std::fputs("\n", stderr);
  return 0;
}

}  // namespace
```
