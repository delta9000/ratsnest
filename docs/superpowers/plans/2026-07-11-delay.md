# Delay Module Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement Phase 2's `delay` — a mono tape-style feedback delay with a 10 s buffer, v_oct time modulation, and hot-reload-surviving echo tails — plus FTZ on the offline render path and a Karplus-Strong demo patch.

**Architecture:** One module file per the established pattern (`src/dsp/modules/delay.cpp` + extern + registry entry). Ring buffer allocated in `init()` (control thread), cubic-Hermite fractional read, one-pole time slew, feedback loop with tone lowpass → DC blocker → tanh, equal-power mix. Design rationale: `docs/superpowers/plans/2026-07-11-delay-design.md`.

**Tech Stack:** C++20, doctest, CMake (globs pick up new files). Build/test via `mise run build` / `mise run test`.

## Global Constraints

- `Module::process()` never allocates, locks, or does I/O. `copy_state_from()` is RT-called: same rule — `std::copy` of the ring is bounded and alloc-free, `assign`/`resize` are forbidden there.
- Everything is one signal: floats at audio rate. No normalling; modulation sums onto declared bases; unpatched inputs read defaults — defaults must be behavior-neutral where the spec implies it.
- Signal conventions (spec §4): audio ±1, gates rise through 0.5, `pitch` is v_oct octaves, time in seconds (`ms` sugar).
- Migration tests must be **load-bearing**: deleting any line of `copy_state_from` must fail a test (mutation-verify).
- The offline render path and tests run WITHOUT FTZ until Task 3 — the module's own denormal flush is what keeps `render` and `play` identical.
- Docs are part of the diff: the spec's Phase 2 delay row is updated in Task 1's commit.
- External deps stay PipeWire + doctest. Conventional commits, no AI attribution.

**Reference implementations:** `src/dsp/modules/svf.cpp` (filter + clamping idiom), `func.cpp` (seconds→samples), `lpg.cpp` (one-pole slews), `quant.cpp` (init-time vector, RT-read-only), `xfade.cpp` (equal-power mix). Engine contract: per-sample `process(ins, outs)`; `init()` on control thread; `copy_state_from()` on the audio thread.

#include <doctest/doctest.h>
#include <cmath>
#include "dsp/modules.h"

using namespace ms::dsp;

namespace {
struct DelayRig {
  std::unique_ptr<Module> module;
  // in, time, pitch, feedback, mix, tone, slew
  float ins[7] = {0.0f, 0.1f, 0.0f, 0.0f, 1.0f, 8000.0f, 0.0f};

  DelayRig() {
    const ModuleInfo* info = find_module("delay");
    REQUIRE(info != nullptr);
    module = info->create();
    module->init(48000.0, {});
  }

  float tick(float in) {
    ins[0] = in;
    float out[1];
    module->process(ins, out);
    return out[0];
  }
};

// Peak magnitude and its index over [from, from+len).
struct Peak { float value; int at; };
Peak peak_in(DelayRig& d, int from, int len, int now) {
  Peak p{0.0f, -1};
  for (int i = now; i < from + len; ++i) {
    const float o = d.tick(0.0f);
    if (i >= from && std::abs(o) > std::abs(p.value)) p = {o, i};
  }
  return p;
}
}  // namespace

TEST_CASE("delay echoes an impulse after exactly time seconds") {
  DelayRig d;  // time 0.1 s, wet-only, no feedback, slew 0
  (void)d.tick(1.0f);
  const Peak p = peak_in(d, 4700, 200, 1);
  CHECK(p.at == 4800);
  CHECK(p.value == doctest::Approx(1.0f));
}

TEST_CASE("pitch input scales time exponentially, one octave per unit") {
  DelayRig d;
  d.ins[1] = 0.01f;  // 480 samples at pitch 0
  d.ins[2] = 1.0f;   // one octave up: halves to 240
  (void)d.tick(1.0f);
  const Peak p = peak_in(d, 200, 100, 1);
  CHECK(p.at == 240);
  CHECK(p.value == doctest::Approx(1.0f));
}

TEST_CASE("feedback repeats decay through the loop") {
  DelayRig d;
  d.ins[3] = 0.5f;
  d.ins[5] = 20000.0f;  // bright loop so the 2nd peak stays sharp
  (void)d.tick(1.0f);
  const Peak first = peak_in(d, 4700, 200, 1);
  const Peak second = peak_in(d, 9500, 300, 4901);
  CHECK(first.value == doctest::Approx(1.0f));
  CHECK(std::abs(second.value) > 0.3f);
  CHECK(std::abs(second.value) < 0.6f);  // ~tanh(0.5·~0.93) ≈ 0.43
}

TEST_CASE("mix is equal-power: dry at zero, minus-3dB at half") {
  DelayRig dry;
  dry.ins[4] = 0.0f;
  for (int i = 0; i < 100; ++i) {
    const float x = std::sin(0.05f * static_cast<float>(i));
    CHECK(dry.tick(x) == doctest::Approx(x));  // mix 0 == pure dry
  }
  DelayRig half;
  half.ins[4] = 0.5f;
  float last = 0.0f;
  for (int i = 0; i < 100; ++i) last = half.tick(1.0f);  // wet still silent
  CHECK(last == doctest::Approx(std::sqrt(0.5f)).epsilon(0.001));
}

TEST_CASE("slewed time changes keep the output continuous") {
  DelayRig d;
  d.ins[6] = 0.05f;  // 50 ms tape slew
  float prev = 0.0f;
  float max_jump = 0.0f;
  for (int i = 0; i < 24000; ++i) {
    if (i == 12000) d.ins[1] = 0.2f;  // yank the time knob mid-flight
    const float o = d.tick(std::sin(0.13f * static_cast<float>(i)));
    if (i > 4800) max_jump = std::max(max_jump, std::abs(o - prev));
    prev = o;
  }
  CHECK(max_jump < 0.3f);  // doppler bend, not a splice
}

TEST_CASE("tone darkens the repeats, not the first arrival's timing") {
  DelayRig darkrig;
  DelayRig brightrig;
  darkrig.ins[3] = brightrig.ins[3] = 0.7f;
  darkrig.ins[5] = 500.0f;
  brightrig.ins[5] = 12000.0f;
  (void)darkrig.tick(1.0f);
  (void)brightrig.tick(1.0f);
  const Peak dark2 = peak_in(darkrig, 9500, 300, 1);
  const Peak bright2 = peak_in(brightrig, 9500, 300, 1);
  CHECK(std::abs(dark2.value) < std::abs(bright2.value) * 0.6f);
}

TEST_CASE("feedback past unity saturates into sustained oscillation") {
  DelayRig d;
  d.ins[1] = 0.05f;
  d.ins[3] = 1.2f;
  d.ins[5] = 4000.0f;
  (void)d.tick(1.0f);
  double tail = 0.0;
  for (int i = 1; i < 144000; ++i) {  // 3 s
    const float o = d.tick(0.0f);
    REQUIRE(std::isfinite(o));
    REQUIRE(std::abs(o) < 3.0f);
    if (i >= 120000) tail += static_cast<double>(o) * o;
  }
  CHECK(std::sqrt(tail / 24000.0) > 0.005);  // still singing in the last 0.5 s
}

TEST_CASE("decayed tails flush to exact silence, no denormal residue") {
  DelayRig d;
  d.ins[3] = 0.5f;
  (void)d.tick(1.0f);
  bool silent = true;
  for (int i = 1; i < 720000; ++i) {  // 15 s of 0.1 s repeats at -7 dB each
    const float o = d.tick(0.0f);
    if (i >= 715200 && o != 0.0f) silent = false;
  }
  CHECK(silent);  // the state flush reaches exact zero, render == play
}

```cpp
#include <algorithm>
#include <cmath>
#include <cstddef>
#include <numbers>
#include <vector>
#include "dsp/modules.h"

namespace ms::dsp {
namespace {

constexpr InputInfo kIns[] = {{"in", 0.f},        {"time", 0.4f},
                              {"pitch", 0.f},     {"feedback", 0.35f},
                              {"mix", 0.5f},      {"tone", 8000.f},
                              {"slew", 0.1f}};
constexpr OutputInfo kOuts[] = {{"out"}};
constexpr float kMaxSeconds = 10.0f;

class Delay final : public Module {
 public:
  void init(double rate, const Config&) override {
    rate_ = static_cast<float>(rate);
    const auto need = static_cast<std::size_t>(kMaxSeconds * rate) + 4;
    std::size_t size = 1;
    while (size < need) size <<= 1;
    buf_.assign(size, 0.0f);  // control thread: allocation is legal here
    mask_ = size - 1;
    write_ = 0;
    delay_ = -1.0;  // sentinel: snap to the target on the first sample
    lp_ = 0.0f;
    dc_x_ = 0.0f;
    dc_y_ = 0.0f;
  }

  void process(const float* ins, float* outs) override {
    const float secs =
        std::clamp(ins[1], 0.0f, kMaxSeconds) * std::exp2(-ins[2]);
    const double target =
        std::clamp(static_cast<double>(secs) * static_cast<double>(rate_),
                   2.0, static_cast<double>(buf_.size() - 4));
    if (delay_ < 0.0) delay_ = target;
    const float slew_samples = std::max(ins[6], 0.0f) * rate_;
    const double a =
        slew_samples <= 1.0f
            ? 1.0
            : 1.0 - std::exp(-1.0 / static_cast<double>(slew_samples));
    delay_ += (target - delay_) * a;

    // 4-point cubic Hermite read: f interpolates tap di toward tap di+1.
    const auto di = static_cast<std::size_t>(delay_);
    const auto f = static_cast<float>(delay_ - static_cast<double>(di));
    const std::size_t w = write_;
    const float x0 = buf_[(w - (di - 1)) & mask_];
    const float x1 = buf_[(w - di) & mask_];
    const float x2 = buf_[(w - (di + 1)) & mask_];
    const float x3 = buf_[(w - (di + 2)) & mask_];
    const float c1 = 0.5f * (x2 - x0);
    const float c2 = x0 - 2.5f * x1 + 2.0f * x2 - 0.5f * x3;
    const float c3 = 0.5f * (x3 - x0) + 1.5f * (x1 - x2);
    const float wet = ((c3 * f + c2) * f + c1) * f + x1;

    // Feedback path: darken, block DC, saturate, write.
    const float fc = std::clamp(ins[5], 20.0f, 0.45f * rate_);
    const float g =
        1.0f - std::exp(-2.0f * std::numbers::pi_v<float> * fc / rate_);
    lp_ += g * (wet - lp_);
    const float rpole =
        1.0f - 2.0f * std::numbers::pi_v<float> * 10.0f / rate_;
    const float dc = lp_ - dc_x_ + rpole * dc_y_;
    dc_x_ = lp_;
    dc_y_ = dc;
    // The offline render path has no FTZ; flush decaying states ourselves so
    // render and play behave identically.
    if (std::abs(lp_) < 1e-20f) lp_ = 0.0f;
    if (std::abs(dc_y_) < 1e-20f) dc_y_ = 0.0f;
    const float fb = std::clamp(ins[3], 0.0f, 1.2f);
    buf_[w] = ins[0] + std::tanh(fb * dc);
    write_ = (w + 1) & mask_;

    const float m = std::clamp(ins[4], 0.0f, 1.0f);
    const float angle = m * (std::numbers::pi_v<float> / 2.0f);
    outs[0] = ins[0] * std::cos(angle) + wet * std::sin(angle);
  }

  void copy_state_from(const Module& other) override {
    const auto& o = static_cast<const Delay&>(other);
    // RT-called: bounded copy, no allocation. Sizes match in practice (same
    // rate); min() is defense if they ever don't.
    const std::size_t n = std::min(buf_.size(), o.buf_.size());
    std::copy_n(o.buf_.begin(), n, buf_.begin());
    write_ = o.write_ & mask_;
    delay_ = std::min(o.delay_, static_cast<double>(buf_.size() - 4));
    lp_ = o.lp_;
    dc_x_ = o.dc_x_;
    dc_y_ = o.dc_y_;
  }

 private:
  float rate_ = 48000.0f;
  std::vector<float> buf_;
  std::size_t mask_ = 0;
  std::size_t write_ = 0;
  double delay_ = -1.0;
  float lp_ = 0.0f;
  float dc_x_ = 0.0f;
  float dc_y_ = 0.0f;
};

std::unique_ptr<Module> make() { return std::make_unique<Delay>(); }
}  // namespace

const ModuleInfo delay_info{"delay", kIns, {}, kOuts, &make};
}  // namespace ms::dsp
```

Add the extern to `modules.h`, `&delay_info` to `kRegistry`, and the spec row edit listed under Files.

// The copy must carry the ring buffer itself (echoes in flight), the write
// head, the slewed delay length, and the feedback-filter states. A short
// delay with feedback keeps energy circulating through ALL of that state at
// the copy point; comparing across the next echo arrival makes each copied
// member load-bearing.
TEST_CASE("delay migrates ring buffer, write head, and loop filter state") {
  const ModuleInfo* info = find_module("delay");
  REQUIRE(info != nullptr);
  auto a = info->create();
  auto b = info->create();
  a->init(48000.0, {});
  b->init(48000.0, {});
  // in, time, pitch, feedback, mix, tone, slew
  float ins[7] = {0.0f, 0.005f, 0.0f, 0.8f, 1.0f, 3000.0f, 0.0f};
  float out_a[1], out_b[1];
  ins[0] = 1.0f;
  a->process(ins, out_a);
  ins[0] = 0.0f;
  for (int i = 1; i < 700; ++i) {
    ins[0] = i % 190 == 0 ? 0.7f : 0.0f;  // keep fresh energy circulating
    a->process(ins, out_a);
  }
  b->copy_state_from(*a);
  for (int i = 700; i < 1400; ++i) {  // crosses several 240-sample echoes
    ins[0] = i % 190 == 0 ? 0.7f : 0.0f;
    a->process(ins, out_a);
    b->process(ins, out_b);
    CHECK(out_a[0] == doctest::Approx(out_b[0]));
  }
}

# pluck -- Karplus-Strong voice: a 12 ms noise burst rings a tuned delay.
# The delay's pitch input is v_oct, so the quantized melody keeps the string
# in tune with itself; feedback 0.96 sets the decay, tone 3500 the damping.
# (The loop lowpass flattens the pitch slightly, as real strings do.)

clk = clock { bpm: 180 }

# -- exciter: short white-noise burst per beat --
burst = func { rise: 0.001, fall: 0.012, curve: exp }
clk -> burst.trig
hiss = noise { color: white, seed: 4 }
exciter = vca { gain: 0 }
hiss -> exciter.in
burst -> exciter.cv

# -- melody: wandering voltage, sampled per beat, major pentatonic --
wander = random { rate: 2.3, smooth: 0, seed: 6 }
notes = sh { mode: sample }
wander -> notes.in
clk -> notes.trig
scale = quant { scale: [0, 2, 4, 7, 9] }
notes -> scale.in : 0.3

# -- the string: 9.09 ms loop is roughly A2 (110 Hz); pitch tracks the melody --
string = delay { time: 9.09ms, pitch: 0, feedback: 0.96, tone: 3500, mix: 1, slew: 0 }
exciter -> string.in
scale -> string.pitch

# -- space: a gentle echo behind the plucks --
room = delay { time: 370ms, feedback: 0.45, tone: 4500, mix: 0.35 }
string -> room.in : 0.8

room -> out.in : 0.8
-0.1 -> out.level
